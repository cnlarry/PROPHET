```cpp
std::vector<Instruction> BytecodeCompiler::compile(const ASTNodePtr& condition) {
    code_.clear();
    compile_node(condition);
    emit(Opcode::RETURN);
    return code_;
}

void BytecodeCompiler::compile_node(const ASTNodePtr& node) {
    switch (node->type) {
        case ASTNodeType::BINARY_OP:
            compile_binary_op(static_cast<BinaryOpNode*>(node.get()));
            break;
        case ASTNodeType::INDICATOR_REF:
            compile_indicator_ref(static_cast<IndicatorRefNode*>(node.get()));
            break;
        // ... 其他节点类型
    }
}

void BytecodeCompiler::compile_binary_op(const BinaryOpNode* node) {
    // 编译左右操作数
    compile_node(node->left);
    compile_node(node->right);
    
    // 生成运算指令
    switch (node->op) {
        case TokenType::PLUS: emit(Opcode::ADD); break;
        case TokenType::MINUS: emit(Opcode::SUB); break;
        case TokenType::GT: emit(Opcode::GT); break;
        // ...
    }
}
```

**需要实现的方法**:
- `compile_node()` - 节点分发
- `compile_binary_op()` - 二元运算
- `compile_unary_op()` - 一元运算
- `compile_indicator_ref()` - 指标引用
- `compile_env_var()` - 环境变量
- `compile_number()` - 数字常量
- `compile_boolean()` - 布尔常量
- `compile_signal_function()` - ALL/ANY/NONE
- `emit()` - 生成指令

```cpp
Value BytecodeVM::execute(const std::vector<Instruction>& code,
                          dsl::Context& context) {
    stack_.clear();
    ip_ = 0;
    
    while (ip_ < code.size()) {
        const auto& inst = code[ip_];
        
        switch (inst.opcode) {
            case Opcode::LOAD_CONST:
                push(inst.operand);
                break;
                
            case Opcode::LOAD_INDICATOR: {
                // 从operand中解析timeframe, indicator, field
                // 调用context.getOrCalculateIndicator()
                // 提取字段值，压栈
                break;
            }
            
            case Opcode::ADD: {
                auto b = pop();
                auto a = pop();
                push(Value(a.toNumber() + b.toNumber()));
                break;
            }
            
            // ... 其他指令
            
            case Opcode::RETURN:
                return pop();
        }
        
        ip_++;
    }
    
    return Value::undefined();
}
```

## 🔧 实施注意事项

### 1. 指标引用的处理

**挑战**: LOAD_INDICATOR需要从operand解析出3个字符串
```cpp
// 建议方案：使用JSON格式存储
Value operand = Value::fromString("{\"tf\":\"5m\",\"ind\":\"RSI\",\"field\":\"value\"}");

// 或使用特殊分隔符
Value operand = Value::fromString("5m|RSI|value");
```

### 2. 聚合函数（ALL/ANY/NONE）

**挑战**: 需要知道条件数量
```cpp
// 编译时记录
emit(Opcode::ALL, Value::fromNumber(3));  // 3个条件

// 执行时
case Opcode::ALL: {
    int count = inst.operand.toNumber();
    bool result = true;
    for (int i = 0; i < count; i++) {
        result = result && pop().toBool();
    }
    push(Value::fromBool(result));
}
```
