## 🔬 **技术细节**

### **SIMD实现模板**

```cpp
// Prophet.Core/src/simd/indicators/atr.cpp

#include "prophet/simd/indicator.hpp"
#include "prophet/simd/math.hpp"

namespace prophet {
namespace simd {
namespace indicator {

void atr_simd(
    const double* high,
    const double* low,
    const double* close,
    size_t count,
    int PERIOD,
    double* result
) {
    // 1. 计算True Range
    std::vector<double> tr(count);
    
    #ifdef PROPHET_SIMD_AVX2
    // AVX2路径：8个double并行
    for (size_t i = 1; i < count; i += 4) {
        __m256d h = _mm256_loadu_pd(high + i);
        __m256d l = _mm256_loadu_pd(low + i);
        __m256d c_prev = _mm256_loadu_pd(close + i - 1);
        
        // TR = max(H-L, abs(H-C_prev), abs(L-C_prev))
        __m256d hl = _mm256_sub_pd(h, l);
        __m256d hc = _mm256_sub_pd(h, c_prev);
        __m256d lc = _mm256_sub_pd(l, c_prev);
        
        hc = _mm256_andnot_pd(_mm256_set1_pd(-0.0), hc);  // abs
        lc = _mm256_andnot_pd(_mm256_set1_pd(-0.0), lc);  // abs
        
        __m256d tr_val = _mm256_max_pd(hl, hc);
        tr_val = _mm256_max_pd(tr_val, lc);
        
        _mm256_storeu_pd(tr.data() + i, tr_val);
    }
    #else
    // 标量路径
    for (size_t i = 1; i < count; ++i) {
        double hl = high[i] - low[i];
        double hc = std::abs(high[i] - close[i-1]);
        double lc = std::abs(low[i] - close[i-1]);
        tr[i] = std::max({hl, hc, lc});
    }
    #endif
    
    // 2. 对TR应用EMA
    ema_simd(tr.data(), count, PERIOD, result);
}

} // namespace indicator
} // namespace simd
} // namespace prophet
```

# examples/simd_batch_performance_test.py

import time
import numpy as np
from Prophet import Core, Indicator

def benchmark_indicator(name: str, func, args, count=1000):
    """性能测试单个指标"""
    times = []
    for _ in range(count):
        start = time.perf_counter()
        result = func(*args)
        times.append(time.perf_counter() - start)
    
    median = np.median(times) * 1e6  # μs
    return {
        'name': name,
        'median_us': median,
        'throughput_ops': 1e6 / median
    }

def main():
    # 生成测试数据

    prices = generate_klines(count)
