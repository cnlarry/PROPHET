### 1. BacktestEngine (回测引擎核心)

#### 类定义

```csharp
namespace Prophet.Client.Backtest.Engine;

public class BacktestEngine : IExecutionEngine
{
    // ========== 依赖注入 ==========
    private readonly IDataFeed _dataFeed;
    private readonly IOrderManager _orderManager;
    private readonly IPerformanceAnalyzer _performanceAnalyzer;
    private readonly IBacktestStorage _storage;
    private readonly StrategyRunner _strategyRunner;
    
    // ========== 状态 ==========
    private List<Candlestick>? _cachedCandles;
    private bool _isDataPrepared;
    
    // ========== 事件 ==========
    public event EventHandler<BacktestProgressEventArgs>? ProgressChanged;
    public event EventHandler<SignalGeneratedEventArgs>? SignalGenerated;
    public event EventHandler<OrderExecutedEventArgs>? OrderExecuted;
    
    // ========== 性能统计 ==========
    private DateTime _lastProgressTime;
    private int _lastProgressIndex;
    
    // ========== 构造函数 ==========
    public BacktestEngine(
        IDataFeed dataFeed,
        IOrderManager orderManager,
        IPerformanceAnalyzer performanceAnalyzer,
        IBacktestStorage storage)
    {
        _dataFeed = dataFeed ?? throw new ArgumentNullException(nameof(dataFeed));
        _orderManager = orderManager ?? throw new ArgumentNullException(nameof(orderManager));
        _performanceAnalyzer = performanceAnalyzer ?? throw new ArgumentNullException(nameof(performanceAnalyzer));
        _storage = storage ?? throw new ArgumentNullException(nameof(storage));
        _strategyRunner = new StrategyRunner();
    }
    
    // ========== 核心方法 ==========
    
    /// <summary>
    /// 准备数据（预加载K线数据）
    /// 🚀 性能关键：只加载一次！
    /// </summary>
    public async Task PrepareDataAsync(
        string symbol,
        string interval,
        DateTime startDate,
        DateTime endDate,
        CancellationToken cancellationToken = default)
    {
        Console.WriteLine($"📊 开始加载历史数据: {symbol} {interval} {startDate:yyyy-MM-dd} ~ {endDate:yyyy-MM-dd}");
        
        var sw = System.Diagnostics.Stopwatch.StartNew();
        
        _cachedCandles = await _dataFeed.LoadCandlesAsync(
            symbol, 
            interval, 
            startDate, 
            endDate,
            cancellationToken
        );
        
        sw.Stop();
        
        Console.WriteLine($"✅ 数据加载完成: {_cachedCandles.Count} 根K线，耗时 {sw.ElapsedMilliseconds}ms");
        
        _isDataPrepared = true;
    }
    
    /// <summary>
    /// 运行回测
    /// </summary>
    public async Task<BacktestResult> RunAsync(
        Strategy strategy,
        BacktestConfig config,
        CancellationToken cancellationToken = default)
    {
        // 1. 验证
        if (!_isDataPrepared || _cachedCandles == null)
        {
            throw new InvalidOperationException("数据未准备，请先调用 PrepareDataAsync()");
        }
        
        if (strategy == null)
        {
            throw new ArgumentNullException(nameof(strategy));
        }
        
        // 2. 初始化
        var backtestId = Guid.NewGuid().ToString();
        var startTime = DateTime.Now;
        
        _orderManager.Initialize(config.InitialCapital);
        _performanceAnalyzer.Initialize(config);
        
        Console.WriteLine($"🚀 开始回测: {strategy.Name}");
        Console.WriteLine($"   时间范围: {config.StartDate:yyyy-MM-dd} ~ {config.EndDate:yyyy-MM-dd}");
        Console.WriteLine($"   初始资金: {config.InitialCapital:N2}");
        Console.WriteLine($"   K线数量: {_cachedCandles.Count}");
        
        // 3. 主回测循环
        for (int i = 0; i < _cachedCandles.Count; i++)
        {
            // 检查取消
            if (cancellationToken.IsCancellationRequested)
            {
                Console.WriteLine("⚠️ 回测已取消");
                return CreateCancelledResult(backtestId, config);
            }
            
            var candle = _cachedCandles[i];
            
            // 3.1 获取当前时刻的历史数据窗口
            var windowData = _cachedCandles.Take(i + 1).ToList();
            
            // 3.2 调用策略引擎（Prophet.Core）
            var signal = _strategyRunner.Evaluate(windowData, strategy);
            
            // 3.3 处理信号
            if (signal.Action != SignalAction.None)
            {
                // 触发信号事件（UI绘制）
                SignalGenerated?.Invoke(this, new SignalGeneratedEventArgs
                {
                    BacktestId = backtestId,
                    Signal = signal,
                    Candle = candle,
                    CandleIndex = i,
                    GlobalIndex = i // 在_cachedCandles中的全局索引
                });
                
                // 执行订单
                var order = _orderManager.ProcessSignal(signal, candle);
                if (order != null)
                {
                    OrderExecuted?.Invoke(this, new OrderExecutedEventArgs
                    {
                        BacktestId = backtestId,
                        Order = order
                    });
                }
            }
            
            // 3.4 更新绩效统计
            _performanceAnalyzer.Update(candle, _orderManager.CurrentEquity);
            
            // 3.5 更新进度（每100根K线更新一次，减少UI刷新）
            if (i % 100 == 0 || i == _cachedCandles.Count - 1)
            {
                // 计算剩余时间估算
                var estimatedRemaining = EstimateRemainingTime(i, _cachedCandles.Count, startTime);
                
                ProgressChanged?.Invoke(this, new BacktestProgressEventArgs
                {
                    BacktestId = backtestId,
                    Current = i + 1,
                    Total = _cachedCandles.Count,
                    Percentage = (i + 1) * 100.0 / _cachedCandles.Count,
                    CurrentTime = candle.Time,
                    CurrentEquity = _orderManager.CurrentEquity,
                    EstimatedTimeRemaining = estimatedRemaining  // 剩余时间估算
                });
            }
        }
        
        // 4. 生成报告（内存操作，不访问数据库）
        var result = _performanceAnalyzer.GenerateReport();
        result.BacktestId = backtestId;
        result.StrategyId = strategy.Id;
        result.StrategyName = strategy.Name;
        result.Config = config;
        result.StartTime = startTime;
        result.EndTime = DateTime.Now;
        result.Duration = result.EndTime - result.StartTime;
        
        // 收集所有订单和信号（内存中）
        result.Orders = _orderManager.ClosedOrders.ToList();
        result.Signals = CollectAllSignals();  // 收集所有信号事件
        
        // 5. 回测结束后统一落库（批量写入，避免频繁访问数据库）
        await _storage.SaveBacktestResultAsync(result, cancellationToken);
        
        // 6. 输出统计
        PrintSummary(result);
        
        return result;
    }
    
    /// <summary>
    /// 批量回测（参数优化）
    /// 🚀 性能关键：并行执行！
    /// </summary>
    public async Task<List<BacktestResult>> RunBatchAsync(
        Strategy strategy,
        List<BacktestConfig> configs,
        ParallelOptions? parallelOptions = null,
        CancellationToken cancellationToken = default)
    {
        if (!_isDataPrepared)
        {
            throw new InvalidOperationException("数据未准备，请先调用 PrepareDataAsync()");
        }
        
        Console.WriteLine($"🔄 开始批量回测: {configs.Count} 个参数组合");
        Console.WriteLine($"   并行度: {parallelOptions?.MaxDegreeOfParallelism ?? Environment.ProcessorCount}");
        
        var results = new ConcurrentBag<BacktestResult>();
        var completed = 0;
        
        var options = parallelOptions ?? new ParallelOptions
        {
            MaxDegreeOfParallelism = 4, // 默认4线程
            CancellationToken = cancellationToken
        };
        
        await Parallel.ForEachAsync(configs, options, async (config, ct) =>
        {
            // 每个线程创建独立的引擎组件
            var engine = CreateIndependentEngine();
            var result = await engine.RunAsync(strategy, config, ct);
            results.Add(result);
            
            var current = Interlocked.Increment(ref completed);
            Console.WriteLine($"   进度: {current}/{configs.Count} ({current * 100.0 / configs.Count:F1}%)");
        });
        
        Console.WriteLine($"✅ 批量回测完成: {results.Count} 个结果");
        
        return results.OrderByDescending(r => r.TotalReturn).ToList();
    }
    
    // ========== 私有方法 ==========
    
    private BacktestEngine CreateIndependentEngine()
    {
        // 创建线程安全的独立引擎实例
        return new BacktestEngine(
            _dataFeed,
            new SimulatedOrderManager(),
            new PerformanceAnalyzer(),
            _storage
        )
        {
            _cachedCandles = this._cachedCandles, // 共享缓存数据
            _isDataPrepared = true
        };
    }
    
    private BacktestResult CreateCancelledResult(string backtestId, BacktestConfig config)
    {
        return new BacktestResult
        {
            BacktestId = backtestId,
            Config = config,
            Status = BacktestStatus.Cancelled,
            ErrorMessage = "用户取消回测"
        };
    }
    
    private void PrintSummary(BacktestResult result)
    {
        Console.WriteLine($"\n📊 回测结果摘要:");
        Console.WriteLine($"   回测ID: {result.BacktestId}");
        Console.WriteLine($"   耗时: {result.Duration.TotalSeconds:F2}秒");
        Console.WriteLine($"   总收益率: {result.TotalReturn:P2}");
        Console.WriteLine($"   年化收益率: {result.AnnualizedReturn:P2}");
        Console.WriteLine($"   最大回撤: {result.MaxDrawdown:P2}");
        Console.WriteLine($"   夏普比率: {result.SharpeRatio:F2}");
        Console.WriteLine($"   索提诺比率: {result.SortinoRatio:F2}");
        Console.WriteLine($"   胜率: {result.WinRate:P2}");
        Console.WriteLine($"   总交易次数: {result.TotalTrades}");
        Console.WriteLine($"   盈利次数: {result.WinningTrades}");
        Console.WriteLine($"   亏损次数: {result.LosingTrades}");
        Console.WriteLine($"   盈亏比: {result.ProfitFactor:F2}");
        Console.WriteLine($"   最大连续盈利: {result.MaxConsecutiveWins}次");
        Console.WriteLine($"   最大连续亏损: {result.MaxConsecutiveLosses}次");
        Console.WriteLine($"   平均持仓时间: {result.AvgHoldingTime.TotalHours:F1}小时");
    }
    
    /// <summary>
    /// 估算剩余时间
    /// </summary>
    private TimeSpan EstimateRemainingTime(int current, int total, DateTime startTime)
    {
        if (current == 0) return TimeSpan.Zero;
        
        var elapsed = DateTime.Now - startTime;
        var avgTimePerCandle = elapsed.TotalSeconds / current;
        var remaining = (total - current) * avgTimePerCandle;
        
        return TimeSpan.FromSeconds(remaining);
    }
    
    /// <summary>
    /// 收集所有信号事件（包括未成交的）
    /// </summary>
    private List<SignalEvent> CollectAllSignals()
    {
        // 从回测过程中收集的信号列表返回
        // 实现细节：在主循环中记录所有信号
        return new List<SignalEvent>();
    }
}
```

```csharp
namespace Prophet.Client.Backtest.OrderManagement;

public class SimulatedOrderManager : IOrderManager
{
    private decimal _cash;
    private decimal _position;
    private List<Order> _openOrders = new();
    private List<Order> _closedOrders = new();
    
    // ========== 配置 ==========
    public decimal SlippageRate { get; set; } = 0.0005m; // 0.05%
    public decimal TakerFeeRate { get; set; } = 0.001m;  // 0.1%
    
    // ========== 属性 ==========
    public decimal CurrentEquity => _cash + _position;
    public decimal CurrentCash => _cash;
    public decimal CurrentPosition => _position;
    public List<Order> OpenOrders => new List<Order>(_openOrders);
    public List<Order> ClosedOrders => new List<Order>(_closedOrders);
    
    // ========== 事件 ==========
    public event EventHandler<OrderFilledEventArgs>? OrderFilled;
    public event EventHandler<StopLossTriggeredEventArgs>? StopLossTriggered;
    public event EventHandler<TakeProfitTriggeredEventArgs>? TakeProfitTriggered;
    
    // ========== 初始化 ==========
    public void Initialize(decimal initialCapital)
    {
        _cash = initialCapital;
        _position = 0;
        _openOrders.Clear();
        _closedOrders.Clear();
    }
    
    // ========== 处理信号 ==========
    public Order? ProcessSignal(Signal signal, Candlestick candle)
    {
        switch (signal.Action)
        {
            case SignalAction.Buy:
                return ProcessBuySignal(signal, candle);
                
            case SignalAction.Sell:
                return ProcessSellSignal(signal, candle);
                
            case SignalAction.CloseLong:
                return ProcessCloseLongSignal(signal, candle);
                
            case SignalAction.CloseShort:
                return ProcessCloseShortSignal(signal, candle);
                
            default:
                return null;
        }
    }
    
    private Order? ProcessBuySignal(Signal signal, Candlestick candle)
    {
        // 已有仓位，不重复开仓
        if (_position != 0)
        {
            return null;
        }
        
        // 模拟成交价格（考虑滑点）
        var executionPrice = ApplySlippage(candle.Close, SignalAction.Buy);
        
        // 计算数量（使用全部资金）
        var quantity = _cash / executionPrice;
        
        // 计算手续费
        var fee = quantity * executionPrice * TakerFeeRate;
        
        // 创建订单
        var order = new Order
        {
            Id = Guid.NewGuid().ToString(),
            Side = OrderSide.Buy,
            Type = OrderType.Market,
            Quantity = quantity,
            OpenPrice = executionPrice,
            OpenTime = candle.Time,
            Fee = fee,
            Status = OrderStatus.Filled,
            TakeProfit = signal.TakeProfit,
            StopLoss = signal.StopLoss
        };
        
        // 更新仓位
        _position = quantity * executionPrice - fee;
        _cash = 0;
        
        _openOrders.Add(order);
        
        // 触发事件
        OrderFilled?.Invoke(this, new OrderFilledEventArgs { Order = order });
        
        return order;
    }
    
    private Order? ProcessSellSignal(Signal signal, Candlestick candle)
    {
        // 已有仓位，不重复开仓
        if (_position != 0)
        {
            return null;
        }
        
        // 做空逻辑（如果支持）
        // TODO: 实现做空
        
        return null;
    }
    
    private Order? ProcessCloseLongSignal(Signal signal, Candlestick candle)
    {
        // 没有仓位，无法平仓
        if (_position == 0)
        {
            return null;
        }
        
        // 模拟成交价格（考虑滑点）
        var executionPrice = ApplySlippage(candle.Close, SignalAction.Sell);
        
        // 计算盈亏
        var profit = _position - _position / executionPrice * executionPrice;
        var fee = _position * TakerFeeRate;
        
        // 找到对应的开仓订单
        var openOrder = _openOrders.FirstOrDefault();
        if (openOrder != null)
        {
            openOrder.ClosePrice = executionPrice;
            openOrder.CloseTime = candle.Time;
            openOrder.Profit = profit - fee;
            openOrder.Status = OrderStatus.Closed;
            
            _openOrders.Remove(openOrder);
            _closedOrders.Add(openOrder);
        }
        
        // 更新仓位
        _cash = _position - fee;
        _position = 0;
        
        return openOrder;
    }
    
    private Order? ProcessCloseShortSignal(Signal signal, Candlestick candle)
    {
        // TODO: 实现平空仓逻辑
        return null;
    }
    
    // ========== 滑点模拟 ==========
    private decimal ApplySlippage(decimal price, SignalAction action)
    {
        var slippage = price * SlippageRate;
        return action == SignalAction.Buy 
            ? price + slippage 
            : price - slippage;
    }
    
    // ========== 止盈止损更新 ==========
    public void UpdateStopLoss(Order order, decimal newStopLoss)
    {
        order.StopLoss = newStopLoss;
    }
    
    public void UpdateTakeProfit(Order order, decimal newTakeProfit)
    {
        order.TakeProfit = newTakeProfit;
    }
}
```

```csharp
public class BacktestConfig
{

    public decimal MakerFeeRate { get; set; } = 0.0005m;  // 挂单手续费 0.05%（未来支持）

    // ========== 策略参数（JSON序列化）==========
    public Dictionary<string, object> Parameters { get; set; } = new();
    
    // ========== 性能优化配置 ==========
```

```csharp
/// <summary>
/// 回测进度事件参数（用于UI进度展示）
/// </summary>
public class BacktestProgressEventArgs : EventArgs
{
    public string BacktestId { get; set; } = string.Empty;
    public int Current { get; set; }                    // 当前K线索引
    public int Total { get; set; }                      // 总K线数量
    public double Percentage { get; set; }              // 进度百分比 (0-100)
    public DateTime CurrentTime { get; set; }           // 当前回测的时间点
    public decimal CurrentEquity { get; set; }          // 当前权益
    public TimeSpan EstimatedTimeRemaining { get; set; } // 预计剩余时间
    
    /// <summary>
    /// 格式化显示进度信息
    /// </summary>
    public string FormatProgress()
    {
        return $"{Current:N0} / {Total:N0} ({Percentage:F1}%) - " +
               $"当前时间: {CurrentTime:yyyy-MM-dd HH:mm} - " +
               $"权益: ${CurrentEquity:N2} - " +
               $"剩余: {FormatTimeSpan(EstimatedTimeRemaining)}";
    }
    
    private string FormatTimeSpan(TimeSpan time)
    {
        if (time.TotalMinutes < 1)
            return $"{time.TotalSeconds:F0}秒";
        else if (time.TotalHours < 1)
            return $"{time.TotalMinutes:F0}分钟";
        else
            return $"{time.TotalHours:F1}小时";
    }
}
```

## 🚀 性能优化设计

### K线转换缓存机制

**问题**：Prophet.Core 接收1m K线数据，需要转换成多个时间框架（5m, 15m, 1h等），这个转换过程会消耗大量性能。

**解决方案**：
1. **LRU缓存**：缓存最近转换的K线数据，避免重复计算
2. **向量化（SIMD）**：使用AVX2指令集加速批量计算（max, min, sum）
3. **多线程**：并行转换多个时间框架

**C++端实现**（见独立优化文档）:
- 缓存大小：可配置，默认100次转换结果
- 缓存命中率目标：> 80%
- 性能提升预期：20-100倍（缓存命中时）

**C#端配置**：

```csharp
public class BacktestConfig
{
    // 启用K线转换缓存
    public bool EnableKlineCache { get; set; } = true;
    
    // 缓存大小（保留最近N次转换结果）
    public int MaxCacheSize { get; set; } = 100;
}
```
